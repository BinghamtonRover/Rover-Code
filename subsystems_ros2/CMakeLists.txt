# Find Protobuf C++ library. Allow user overrides; otherwise search common
# system paths. The multiarch library path is resolved via the GNU install
# dirs so this also works on arm64/aarch64 (Raspberry Pi OS, Ubuntu arm64).
if(NOT Protobuf_LIBRARIES OR NOT Protobuf_INCLUDE_DIRS)
  include(GNUInstallDirs)
  get_filename_component(_multiarch_libdir
    "${CMAKE_INSTALL_FULL_LIBDIR}/../lib/${CMAKE_LIBRARY_ARCHITECTURE}"
    ABSOLUTE)
  find_library(Protobuf_LIBRARIES
    NAMES protobuf protobufd
    PATHS
      ${CMAKE_INSTALL_FULL_LIBDIR}
      ${_multiarch_libdir}
      /usr/lib
      /usr/local/lib
  )
  find_path(Protobuf_INCLUDE_DIRS
    google/protobuf/message.h
    PATHS /usr/include /usr/local/include
  )
  if(NOT Protobuf_LIBRARIES OR NOT Protobuf_INCLUDE_DIRS)
    message(FATAL_ERROR "Could not find Protobuf C++ library. Install libprotobuf-dev or set Protobuf_LIBRARIES and Protobuf_INCLUDE_DIRS.")
  endif()
endif()

# ---------------------------------------------------------------------------
# Generated protobuf sources
#
# The sources checked into include/${PROJECT_NAME}/generated were produced by
# protoc 3.12 (Ubuntu 20.04/22.04). Newer protoc versions (e.g. 3.21 on
# Raspberry Pi OS Bookworm) reject those files at compile time. When the
# installed libprotobuf is newer than the one that produced the checked-in
# files, regenerate the sources from the .proto files in burt_network/Protobuf
# so the package compiles on both old and new toolchains.
# ---------------------------------------------------------------------------
set(PROTOBUF_GENERATED_DIR "${CMAKE_CURRENT_SOURCE_DIR}/include/${PROJECT_NAME}/generated")
file(GLOB PROTO_SRCS "${PROTOBUF_GENERATED_DIR}/*.pb.cc")
file(GLOB PROTO_HDRS "${PROTOBUF_GENERATED_DIR}/*.pb.h")

find_package(Protobuf QUIET)
set(_REGEN_STAMP FALSE)
if(Protobuf_FOUND AND Protobuf_PROTOC_EXECUTABLE AND PROTO_SRCS)
  # The "#if PROTOBUF_VERSION < NNNNNNN" guard in every generated header encodes
  # the protoc version that produced it (e.g. 3012004 == 3.12.4).
  set(_checked_in_protoc_num 0)
  file(STRINGS "${PROTOBUF_GENERATED_DIR}/video.pb.h" _pb_version_line
    REGEX "#if PROTOBUF_VERSION < [0-9]+")
  if(_pb_version_line MATCHES "#if PROTOBUF_VERSION < ([0-9]+)")
    set(_checked_in_protoc_num "${CMAKE_MATCH_1}")
  endif()

  # The minimum protoc version accepted by the installed libprotobuf, encoded
  # the same way, lives in port_def.inc (e.g. 3021000 for protobuf 3.21).
  set(_min_protoc_num 0)
  if(EXISTS "${Protobuf_INCLUDE_DIRS}/google/protobuf/port_def.inc")
    file(STRINGS "${Protobuf_INCLUDE_DIRS}/google/protobuf/port_def.inc" _min_line
      REGEX "#define PROTOBUF_MIN_PROTOC_VERSION [0-9]+")
    if(_min_line MATCHES "#define PROTOBUF_MIN_PROTOC_VERSION ([0-9]+)")
      set(_min_protoc_num "${CMAKE_MATCH_1}")
    endif()
  endif()

  if(_min_protoc_num AND _checked_in_protoc_num AND
     _checked_in_protoc_num LESS _min_protoc_num)
    # Locate the .proto sources (submodule burt_network/Protobuf).
    set(_PROTO_SRC_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../burt_network/Protobuf")
    if(EXISTS "${_PROTO_SRC_DIR}/video.proto")
      file(GLOB _PROTO_FILES "${_PROTO_SRC_DIR}/*.proto")
      set(_GEN_OUT "${CMAKE_CURRENT_BINARY_DIR}/generated_protobuf")
      file(MAKE_DIRECTORY "${_GEN_OUT}")
      set(_REGEN_OUTPUTS ${PROTO_SRCS})
      add_custom_command(
        OUTPUT ${_REGEN_OUTPUTS}
        COMMAND "${Protobuf_PROTOC_EXECUTABLE}"
                "--proto_path=${_PROTO_SRC_DIR}"
                "--cpp_out=${_GEN_OUT}"
                ${_PROTO_FILES}
        DEPENDS ${_PROTO_FILES}
        COMMENT "Regenerating protobuf sources (checked-in files were made by an older protoc)"
        VERBATIM
      )
      set(_REGEN_STAMP TRUE)
      set_source_files_properties(${PROTO_SRCS} ${PROTO_HDRS} PROPERTIES GENERATED TRUE)
      message(STATUS "${PROJECT_NAME}: regenerating protobufs with ${Protobuf_PROTOC_EXECUTABLE}")
    else()
      message(WARNING
        "${PROJECT_NAME}: checked-in protobufs were generated by protoc ${_checked_in_protoc_num} "
        "but the installed libprotobuf requires at least ${_min_protoc_num}. Regeneration is required, "
        "but the .proto sources were not found at ${_PROTO_SRC_DIR}. "
        "Run 'git submodule update --init' and rebuild, or regenerate manually.")
    endif()
  endif()
endif()

  DESTINATION lib
)

# Helper macro to build a node executable
macro(add_subsystems_node NODE_NAME)
  add_executable(${NODE_NAME} src/${NODE_NAME}.cpp)
  ament_target_dependencies(${NODE_NAME} rclcpp std_msgs)
  target_link_libraries(${NODE_NAME}
    ${PROJECT_NAME}_utils
    ${Boost_LIBRARIES}
    ${Protobuf_LIBRARIES}
    Threads::Threads
  )
  if(cpp_typesupport_target)
    target_link_libraries(${NODE_NAME} "${cpp_typesupport_target}")
  endif()
  install(TARGETS ${NODE_NAME}
    DESTINATION lib/${PROJECT_NAME}
  )
endmacro()

add_subsystems_node(udp_bridge)
add_subsystems_node(firmware_manager)
add_subsystems_node(gps_node)
add_subsystems_node(imu_node)
add_subsystems_node(subsystems_node)

# Install generated protobuf headers so downstream packages can use them.
# When the sources were regenerated into the build tree, install those instead
# of the (incompatible) checked-in copies.
if(_REGEN_STAMP)
  install(DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated_protobuf/"
    DESTINATION include/${PROJECT_NAME}/generated
    FILES_MATCHING PATTERN "*.pb.h")
else()
  install(DIRECTORY include/${PROJECT_NAME}/generated
    DESTINATION include/${PROJECT_NAME})
endif()
